**Code Example: curriculum_loader.py**
```python
import sqlite3
import json
from pathlib import Path

class CurriculumLoader:
    def __init__(self, db_path=None, json_path=None):
        self.db_path = db_path or "CURRICULUM_DATA/ontario_curriculum.sqlite"
        self.json_path = json_path or "CURRICULUM_DATA/ontario_curriculum_full.json"
    
    def get_expectations_for_course(self, course_code):
        """Retrieve all expectations for a specific Ontario course."""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        query = """
        SELECT e.id, e.text, e.irt_b, e.irt_a, e.irt_c, s.name as strand, t.tag
        FROM Expectation e
        JOIN Strand s ON e.strand_id = s.id
        LEFT JOIN Tag t ON e.id = t.expectation_id
        WHERE e.course_id = ?
        """
        
        cursor.execute(query, [course_code])
        results = {}
        
        for row in cursor.fetchall():
            exp_id, text, irt_b, irt_a, irt_c, strand, tag = row
            if exp_id not in results:
                results[exp_id] = {
                    "id": exp_id,
                    "expectation": text,
                    "irt_b": irt_b,
                    "irt_a": irt_a,
                    "irt_c": irt_c,
                    "strand": strand,
                    "tags": []
                }
            if tag:
                results[exp_id]["tags"].append(tag)
        
        conn.close()
        return list(results.values())
    
    def get_all_courses(self):
        """Return list of all available Ontario course codes."""
        with open(self.json_path, 'r') as f:
            data = json.load(f)
        return list(data.get('courses', {}).keys())
```

from sympy import symbols, Eq, solve
import re

class DualEngineOrchestrator:
    def __init__(self):
        self.deterministic_cache = {}
    
    def verify_math_solution(self, problem_type, student_answer, expected_answer):
        """Use SymPy for ground-truth verification."""
        try:
            if problem_type == "linear_equation":
                x = symbols('x')
                student_eq = Eq(eval(student_answer.replace('=', '==')))
                expected_eq = Eq(eval(expected_answer.replace('=', '==')))
                
                student_sol = solve(student_eq, x)
                expected_sol = solve(expected_eq, x)
                
                return {
                    "verified": student_sol == expected_sol,
                    "method": "symPy",
                    "student_solution": student_sol,
                    "expected_solution": expected_sol
                }
        except Exception as e:
            return {"verified": False, "error": str(e)}
    
    def translate_to_student_level(self, verified_truth, student_reading_level):
        """
        Translate deterministic truth into personalized narrative.
        In production, this calls SLM (Phi-3-mini) via NPU.
        """
        # Placeholder: Flesch-Kincaid adjustment logic
        if student_reading_level < 8:
            simplified = self._simplify_language(verified_truth)
        else:
            simplified = verified_truth
        
        return {
            "original": verified_truth,
            "adapted": simplified,
            "reading_level": student_reading_level,
            "modality_options": ["visual", "abstract", "real_world"]
        }
    
    def _simplify_language(self, text):
        """Simplify technical language for younger students."""
        replacements = {
            "calculate": "find",
            "determine": "figure out",
            "approximately": "about",
            "therefore": "so"
        }
        for complex_word, simple_word in replacements.items():
            text = text.replace(complex_word, simple_word)
        return text

**Code Example: attention_monitor.py**
```python
import time
from datetime import datetime, timedelta

class AttentionMonitor:
    def __init__(self):
        self.sessions = {}  # student_id -> session_data
    
    def start_session(self, student_id):
        """Begin tracking focus time for a student."""
        self.sessions[student_id] = {
            "start_time": datetime.now(),
            "focused_seconds": 0,
            "last_check": datetime.now(),
            "break_reminders": []
        }
    
    def update_focus(self, student_id, is_focused=True):
        """Update focus tracking (call periodically)."""
        if student_id not in self.sessions:
            return
        
        session = self.sessions[student_id]
        now = datetime.now()
        elapsed = (now - session["last_check"]).total_seconds()
        
        if is_focused:
            session["focused_seconds"] += elapsed
        
        # Check for 20-minute interval
        if session["focused_seconds"] % 1200 >= elapsed:
            session["break_reminders"].append({
                "timestamp": now.isoformat(),
                "message": "Consider taking a short break!"
            })
        
        session["last_check"] = now
    
    def get_session_summary(self, student_id):
        """Get focus analytics for a session."""
        if student_id not in self.sessions:
            return {"error": "No active session"}
        
        session = self.sessions[student_id]
        total_minutes = session["focused_seconds"] / 60
        
        return {
            "student_id": student_id,
            "total_focus_minutes": round(total_minutes, 1),
            "break_reminders_count": len(session["break_reminders"]),
            "session_start": session["start_time"].isoformat()
        }
```

## 6. File Organization Best Practices

### 6.1 Naming Conventions

- **Agents**: `{role}_agent.py` (e.g., `teacher_agent.py`, `rag_tutor_agent.py`)
- **Schemas**: `{domain}.schema.json` (e.g., `curriculum_expectation.schema.json`)
- **Config**: `{service}_config.yaml` (e.g., `rag_config.yaml`)
- **Runtime**: `{functionality}.py` (e.g., `adaptive_learning.py`)
- **Adapters**: `{target}_adapter.py` (e.g., `math_capsule_adapter.py`)

### 6.2 Directory Structure Rationale

```
ontario/
├── CURRICULUM_DATA/      # Static data assets (read-only at runtime)
├── contracts/            # Solidity smart contracts (blockchain layer)
├── agents/               # Autonomous AI agents (business logic)
├── schemas/              # JSON/Cap'n Proto schemas (data validation)
├── config/               # YAML/JSON configuration files
├── runtime/              # Core execution engines
└── adapters/             # External capsule integration bridges
```

This structure separates:
- **Data** (CURRICULUM_DATA) from **Logic** (agents, runtime)
- **On-chain** (contracts) from **Off-chain** (agents, runtime)
- **Internal** (runtime) from **External** (adapters)
